const STATIC_EMAIL = "support@rdr.company.com";
const STATIC_PASSWORD = "RDR-Support-2026!";
const PAGE_SIZE = 10;
let allRecords = [];
let filteredRecords = [];
let currentPage = 1;

const app = document.querySelector("#app");

function renderLogin(message = "") {
  app.innerHTML = `
    <main class="login-shell">
      <section class="login-card" aria-label="RDR Support Console login">
        <div class="brand-mark">RR</div>
        <h1>RDR Support Console</h1>
        <p>Sign in to monitor Retail Delivery Report transactions.</p>
        <form id="login-form">
          <div class="field"><label for="email">Email</label><input id="email" type="email" autocomplete="username" placeholder="Enter email" required /></div>
          <div class="field"><label for="password">Password</label><input id="password" type="password" autocomplete="current-password" placeholder="Enter password" required /></div>
          <div class="error" role="alert">${message}</div>
          <button class="login-button" type="submit">Sign in</button>
        </form>
        <p class="hint">Demo access: <strong>${STATIC_EMAIL}</strong> / <strong>${STATIC_PASSWORD}</strong></p>
      </section>
    </main>`;
  document.querySelector("#login-form").addEventListener("submit", (event) => {
    event.preventDefault();
    const email = document.querySelector("#email").value.trim();
    const password = document.querySelector("#password").value;
    if (email === STATIC_EMAIL && password === STATIC_PASSWORD) {
      sessionStorage.setItem("rdr-authenticated", "true");
      renderDashboard();
    } else renderLogin("Invalid demo credentials.");
  });
}

async function loadData() {
  const response = await fetch("../grid_data/rdr_grid_static_data.json");
  if (!response.ok) throw new Error("Unable to load transaction data");
  const payload = await response.json();
  return payload.records;
}

function statusClass(status) { return String(status || "").toLowerCase(); }
function formatDate(value) { return value ? new Date(value).toLocaleString("en-US", { month: "short", day: "numeric", year: "numeric", hour: "2-digit", minute: "2-digit" }) : "—"; }
function displayStatus(record) { return record.status || record.transaction_state || "UNKNOWN"; }
function applyFilters() {
  const vin = document.querySelector("#filter-vin")?.value.trim().toLowerCase() || "";
  const dealer = document.querySelector("#filter-dealer")?.value.trim().toLowerCase() || "";
  const rdr = document.querySelector("#filter-rdr")?.value.trim().toLowerCase() || "";
  const status = document.querySelector("#filter-status")?.value || "";
  filteredRecords = allRecords.filter((record) =>
    (!vin || String(record.vin || "").toLowerCase().includes(vin)) &&
    (!dealer || String(record.dealer_id || "").toLowerCase().includes(dealer)) &&
    (!rdr || String(record.rdr_id || "").toLowerCase().includes(rdr)) &&
    (!status || displayStatus(record) === status)
  );
  currentPage = 1;
  renderTable();
}

function renderDashboard() {
  app.innerHTML = `
    <header class="topbar"><div class="brand-mark">RDR</div><div class="title"><h1>RDR Support Console</h1><p>Transaction Monitoring</p></div><div class="header-actions"><span>Support workspace</span><button class="logout" id="logout">Log out</button></div></header>
    <main class="content">
      <form class="filters" id="filters">
        <div class="field"><label for="filter-vin">VIN</label><input id="filter-vin" placeholder="Enter VIN..." /></div>
        <div class="field"><label for="filter-dealer">Dealer ID</label><input id="filter-dealer" placeholder="Enter Dealer ID..." /></div>
        <div class="field"><label for="filter-rdr">RDR ID</label><input id="filter-rdr" placeholder="Enter RDR ID..." /></div>
        <div class="field"><label for="filter-status">Status</label><select id="filter-status"><option value="">All Statuses</option><option>PENDING</option><option>FAILED</option><option>COMPLETED</option><option>REJECTED</option><option>BLOCKED</option><option>RETRYING</option><option>CANCELLED</option></select></div>
        <div class="field"><label for="filter-date">Date Range</label><input id="filter-date" value="Nov 20, 2026 – Nov 23, 2026" readonly /></div>
        <button class="search-button" type="submit"><span class="search-icon">⌕</span> Search</button>
      </form>
      <section class="card"><div class="card-head"><h2>RDR Transactions</h2><div class="metrics"><div class="metric pending"><span>Pending</span><strong id="pending-count">0</strong></div><div class="metric failed"><span>Failed</span><strong id="failed-count">0</strong></div><div class="metric completed"><span>Completed</span><strong id="completed-count">0</strong></div></div></div><div class="table-wrap"><table><thead><tr><th><input id="select-all" type="checkbox" aria-label="Select all visible rows" /></th><th>RDR ID</th><th>VIN</th><th>Dealer ID</th><th>Customer</th><th>Submitted Time</th><th>Status</th><th>Priority</th><th>Retry Count</th><th>Last Updated</th><th>Last Error</th></tr></thead><tbody id="transaction-rows"></tbody></table></div><div class="table-foot"><span id="result-count"></span><div class="pager" id="pager"></div></div></section>
    </main>`;
  document.querySelector("#filters").addEventListener("submit", (event) => { event.preventDefault(); applyFilters(); });
  // Filter live as well, so a value typed/filled without pressing Search still narrows the grid.
  document.querySelectorAll("#filters input, #filters select").forEach((el) => {
    el.addEventListener("input", applyFilters);
    el.addEventListener("change", applyFilters);
  });
  document.querySelector("#logout").addEventListener("click", () => { sessionStorage.removeItem("rdr-authenticated"); renderLogin(); });
  document.querySelector("#select-all").addEventListener("change", (event) => document.querySelectorAll(".row-check").forEach((box) => { box.checked = event.target.checked; box.closest("tr").classList.toggle("selected", box.checked); }));
  applyFilters();
}

function renderTable() {
  const pending = allRecords.filter((r) => displayStatus(r) === "PENDING").length;
  const failed = allRecords.filter((r) => ["FAILED", "REJECTED", "BLOCKED"].includes(displayStatus(r))).length;
  const completed = allRecords.filter((r) => displayStatus(r) === "COMPLETED").length;
  document.querySelector("#pending-count").textContent = pending;
  document.querySelector("#failed-count").textContent = failed;
  document.querySelector("#completed-count").textContent = completed;
  const pages = Math.max(1, Math.ceil(filteredRecords.length / PAGE_SIZE));
  currentPage = Math.min(currentPage, pages);
  const start = (currentPage - 1) * PAGE_SIZE;
  const rows = filteredRecords.slice(start, start + PAGE_SIZE);
  document.querySelector("#transaction-rows").innerHTML = rows.length ? rows.map((record) => {
    const status = displayStatus(record);
    return `<tr><td><input class="row-check" type="checkbox" aria-label="Select ${record.rdr_id || record.submission_id}" /></td><td>${record.rdr_id ? `<button class="rdr-link" title="View ${record.rdr_id}">${record.rdr_id}</button>` : "—"}</td><td>${record.vin || "—"}</td><td>${record.dealer_id || "—"}</td><td>${record.customer || "—"}</td><td>${formatDate(record.submitted_at)}</td><td><span class="status ${statusClass(status)}">${status}</span></td><td>${record.priority || "—"}</td><td>${record.retry_count ?? 0}</td><td>${formatDate(record.last_updated)}</td><td>${record.last_error_display || "—"}</td></tr>`;
  }).join("") : `<tr><td class="empty" colspan="11">No transactions match the selected filters.</td></tr>`;
  document.querySelectorAll(".row-check").forEach((box) => box.addEventListener("change", (event) => event.target.closest("tr").classList.toggle("selected", event.target.checked)));
  // Make the filtered result obvious: scroll the table into view and mark narrowed results.
  const anyFilter = ["#filter-vin", "#filter-dealer", "#filter-rdr", "#filter-status"].some((id) => document.querySelector(id)?.value);
  document.querySelector("#transaction-rows").classList.toggle("filtered", anyFilter);
  if (anyFilter && rows.length) document.querySelector("#transaction-rows").scrollIntoView({ block: "center", behavior: "smooth" });
  document.querySelector("#result-count").textContent = `Showing ${filteredRecords.length ? start + 1 : 0}–${Math.min(start + PAGE_SIZE, filteredRecords.length)} of ${filteredRecords.length} results`;
  document.querySelector("#pager").innerHTML = `<button aria-label="Previous page" ${currentPage === 1 ? "disabled" : ""} data-page="${currentPage - 1}">‹</button>${Array.from({ length: Math.min(pages, 5) }, (_, index) => `<button class="${currentPage === index + 1 ? "active" : ""}" data-page="${index + 1}">${index + 1}</button>`).join("")}${pages > 5 ? "<span>…</span>" : ""}<button aria-label="Next page" ${currentPage === pages ? "disabled" : ""} data-page="${currentPage + 1}">›</button>`;
  document.querySelectorAll("#pager [data-page]").forEach((button) => button.addEventListener("click", () => { currentPage = Number(button.dataset.page); renderTable(); }));
}

(async function start() {
  try { allRecords = await loadData(); } catch (error) { console.error(error); }
  if (sessionStorage.getItem("rdr-authenticated") === "true") renderDashboard(); else renderLogin();
})();
